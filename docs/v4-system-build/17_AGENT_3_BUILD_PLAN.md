#### Task 3.0.2 — Create `src/db.py` — Shared DB Connection Helper

```python
"""
sentinel/src/db.py
==================
Shared psycopg2 connection context manager.
All writes use ON CONFLICT DO NOTHING for idempotency.
"""
import os
from contextlib import contextmanager
import psycopg2
import psycopg2.extras

# Accept both psycopg2-style and JDBC-style DATABASE_URL
# (JDBC: jdbc:postgresql://... → strip prefix)
_raw = os.environ.get(
    "DATABASE_URL",
    "postgresql://sentinel:sentinel@localhost:5432/sentinel"
)
DATABASE_URL = _raw.replace("jdbc:postgresql", "postgresql") if _raw.startswith("jdbc:") else _raw


@contextmanager
def get_connection():
    """Yields a psycopg2 connection that auto-commits or rolls back on error."""
    conn = psycopg2.connect(DATABASE_URL)
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def execute_batch(conn, sql: str, rows: list[tuple]) -> int:
    """Execute a parameterized INSERT for a batch of rows. Returns row count."""
    if not rows:
        return 0
    with conn.cursor() as cur:
        psycopg2.extras.execute_batch(cur, sql, rows, page_size=100)
    return len(rows)
```

```python
"""
sentinel/src/load.py
====================
V4: Writes ETL output directly to PostgreSQL via psycopg2.
No live_batch.json, no predictions_export.json produced.

import logging
import os

from src.db import get_connection, execute_batch

log = logging.getLogger(__name__)


def load_incidents(df, batch_id: str) -> int:
    """Insert trusted/corrected incidents. ON CONFLICT DO NOTHING."""
    if df is None or df.empty:
        return 0

    SQL = """
        INSERT INTO fact_incidents
            (incident_id, site_id, incident_date, severity, description,
             compliance_score, status, decision, decision_reason,
             batch_id, ingestion_timestamp)
        VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
        ON CONFLICT (incident_id) DO NOTHING
    """

    rows = [
        (
            row.incident_id,
            row.site.lower(),          # normalise to lowercase site_id
            row.incident_date,
            row.severity,
            getattr(row, 'description', None),
            getattr(row, 'compliance_score', None),
            getattr(row, 'status', 'Open'),
            row.decision,
            getattr(row, 'decision_reason', None),
            batch_id,
            datetime.utcnow(),
        )
        for row in df.itertuples()
    ]

    with get_connection() as conn:
        count = execute_batch(conn, SQL, rows)

    log.info("load_incidents: inserted %d rows (batch_id=%s)", count, batch_id[:8])
    return count
```

```python
import os
import base64
import tempfile
import requests
import joblib
from pathlib import Path

API_BASE      = os.environ.get("API_BASE", "http://localhost:8080")
SERVICE_TOKEN = os.environ.get("SERVICE_TOKEN", "")

FALLBACK_MODEL_PATH = Path("models/logreg_v1.pkl")


def load_champion_model():
    """
    Load the current champion model from model_registry.
    Falls back to FALLBACK_MODEL_PATH if the API is unreachable.

    Returns: (pipeline, version_string)
    """
    try:
        resp = requests.get(
            f"{API_BASE}/api/ml/model-registry",
            headers={"X-Service-Token": SERVICE_TOKEN},
            timeout=5,
        )
        resp.raise_for_status()
        registry = resp.json()
        champion = next((m for m in registry if m.get("status") == "champion"), None)
        if champion is None:
            raise ValueError("No champion found in registry")

        artifact_path = Path(champion["artifactPath"])

        # If artifact file exists locally, use it
        if artifact_path.exists():
            return joblib.load(artifact_path), champion["version"]

        # Otherwise, try artifact_blob (base64-encoded PKL stored in DB)
        blob = champion.get("artifactBlob")
        if blob:
            with tempfile.NamedTemporaryFile(suffix=".pkl", delete=False) as f:
                f.write(base64.b64decode(blob))
                tmp_path = f.name
            model = joblib.load(tmp_path)
            Path(tmp_path).unlink(missing_ok=True)
            return model, champion["version"]

        raise FileNotFoundError(
            f"Champion artifact not found at {artifact_path} and no blob available"
        )

    except Exception as e:
        import logging
        logging.getLogger(__name__).warning(
            "Could not load champion from registry (%s) — falling back to %s",
            e, FALLBACK_MODEL_PATH
        )
        return joblib.load(FALLBACK_MODEL_PATH), "logreg_v1"
```

### Phase 2 — Implement `src/retrain.py`

#### Task 3.2.1 — Full Implementation

```python
"""
sentinel/src/retrain.py
=======================
Trains a challenger logistic regression model using accumulated model_feedback.

Usage:
    python -m src.retrain                 # triggered by POST /api/ml/trigger-retrain
    python -m src.retrain --dry-run       # validate data without training

Steps:
    1. Fetch feedback from GET /api/ml/feedback-export
    2. Load fact_site_features from PostgreSQL
    3. Merge feedback labels into training data
    4. Train challenger pipeline
    5. Evaluate on holdout set
    6. Save PKL artifact
    7. POST /api/ml/model-registry (register challenger)
    8. POST /api/ml/training-run   (record run)
"""

import argparse
import base64
import json
import logging
import os
import sys
from datetime import datetime, date
from pathlib import Path

import joblib
import numpy as np
import pandas as pd
import requests
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import f1_score, precision_score, recall_score
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler

log = logging.getLogger(__name__)

API_BASE        = os.environ.get("API_BASE", "http://localhost:8080")
SERVICE_TOKEN   = os.environ.get("SERVICE_TOKEN", "")
MODELS_DIR      = Path("models")
MIN_FEEDBACK    = int(os.environ.get("MIN_FEEDBACK_ROWS", "5"))
TRAIN_CUTOFF    = date(2026, 6, 12)

FEATURES = [
    "days_since_last_audit",
    "rejection_rate_7d",
    "rejection_rate_30d",
    "incident_count_30d",
    "incident_severity_score_30d",
    "pressure_anomaly_count_14d",
    "audit_finding_open_count",
]


def _headers():
    return {"X-Service-Token": SERVICE_TOKEN}


def fetch_feedback() -> pd.DataFrame:
    resp = requests.get(
        f"{API_BASE}/api/ml/feedback-export",
        headers=_headers(),
        params={"excludeUncertain": "true"},
        timeout=10,
    )
    resp.raise_for_status()
    data = resp.json()
    if len(data) < MIN_FEEDBACK:
        log.warning(
            "Insufficient feedback (%d rows, minimum %d). Exiting without training.",
            len(data), MIN_FEEDBACK
        )
        sys.exit(0)
    return pd.DataFrame(data)


def load_features_from_db() -> pd.DataFrame:
    from src.db import get_connection
    with get_connection() as conn:
        with conn.cursor() as cur:
            cur.execute("""
                SELECT site_id, as_of_date,
                       days_since_last_audit, rejection_rate_7d,
                       rejection_rate_30d, incident_count_30d,
                       incident_severity_score_30d, pressure_anomaly_count_14d,
                       audit_finding_open_count
                FROM fact_site_features
                ORDER BY as_of_date
            """)
            rows = cur.fetchall()
            cols = [d[0] for d in cur.description]
    return pd.DataFrame(rows, columns=cols)


def merge_feedback(features_df: pd.DataFrame, feedback_df: pd.DataFrame) -> pd.DataFrame:
    """
    Merge feedback labels into the feature dataset.
    'accurate'   → keep original label (no change)
    'inaccurate' → flip the label
    Rows with no feedback retain their original label from the ground truth.
    """
    # The original ground truth label is built in the same way as predict.py
    # (any Critical incident in next 7 days after as_of_date)
    # For now, use the feature dataset's existing 'label' column if present,
    # otherwise set to 0 (conservative default)
    if "label" not in features_df.columns:
        features_df["label"] = 0

    # Match feedback by site_id + nearest as_of_date
    feedback_lookup = {}
    for _, fb_row in feedback_df.iterrows():
        key = fb_row["siteId"]
        feedback_lookup.setdefault(key, []).append(fb_row["rating"])

    def apply_feedback(row):
        site_ratings = feedback_lookup.get(row["site_id"], [])
        inaccurate_count = site_ratings.count("inaccurate")
        accurate_count   = site_ratings.count("accurate")
        if inaccurate_count > accurate_count:
            return 1 - row["label"]  # flip
        return row["label"]

    features_df["label"] = features_df.apply(apply_feedback, axis=1)
    return features_df


def train(enriched: pd.DataFrame):
    train_df = enriched[enriched["as_of_date"] < TRAIN_CUTOFF].copy()
    test_df  = enriched[enriched["as_of_date"] >= TRAIN_CUTOFF].copy()

    if train_df.empty or test_df.empty:
        log.error("Insufficient data for train/test split. Exiting.")
        sys.exit(1)

    X_train = train_df[FEATURES].fillna({"days_since_last_audit": 999})
    y_train = train_df["label"]
    X_test  = test_df[FEATURES].fillna({"days_since_last_audit": 999})
    y_test  = test_df["label"]

    pipeline = Pipeline([
        ("scaler", StandardScaler()),
        ("model",  LogisticRegression(class_weight="balanced", max_iter=1000)),
    ])
    pipeline.fit(X_train, y_train)

    y_pred = pipeline.predict(X_test)
    metrics = {
        "precision": float(precision_score(y_test, y_pred, zero_division=0)),
        "recall":    float(recall_score(y_test, y_pred, zero_division=0)),
        "f1":        float(f1_score(y_test, y_pred, zero_division=0)),
    }
    log.info("Challenger metrics: P=%.3f R=%.3f F1=%.3f", *metrics.values())

    # Feature importance (absolute coefficient, normalised)
    coefs = np.abs(pipeline["model"].coef_[0])
    feature_importance = dict(zip(FEATURES, (coefs / coefs.sum()).round(3).tolist()))

    return pipeline, metrics, feature_importance, len(train_df)


def get_next_version() -> str:
    resp = requests.get(f"{API_BASE}/api/ml/model-registry", headers=_headers(), timeout=5)
    resp.raise_for_status()
    existing = resp.json()
    return f"logreg_v{len(existing) + 1}"


def register(pipeline, version: str, metrics: dict, feature_importance: dict,
             n_rows: int, n_feedback: int) -> str:
    MODELS_DIR.mkdir(exist_ok=True)
    artifact_path = MODELS_DIR / f"{version}.pkl"
    joblib.dump(pipeline, artifact_path)

    # Base64-encode for DB storage (ephemeral filesystem safety)
    with open(artifact_path, "rb") as f:
        artifact_blob = base64.b64encode(f.read()).decode("utf-8")

    payload = {
        "version":          version,
        "algorithm":        "logistic_regression",
        "precisionScore":   metrics["precision"],
        "recallScore":      metrics["recall"],
        "f1Score":          metrics["f1"],
        "artifactPath":     str(artifact_path),
        "artifactBlob":     artifact_blob,
        "featureImportance": json.dumps(feature_importance),
        "notes":            f"Trained on {n_rows} rows + {n_feedback} feedback rows",
    }
    resp = requests.post(
        f"{API_BASE}/api/ml/model-registry",
        json=payload, headers=_headers(), timeout=10
    )
    resp.raise_for_status()
    return resp.json()["id"]


def record_run(model_id: str, n_rows: int, n_feedback: int, started_at: str) -> None:
    payload = {
        "modelRegistryId":  model_id,
        "triggeredBy":      "manual",
        "rowsUsed":         n_rows,
        "feedbackRowsUsed": n_feedback,
        "startedAt":        started_at,
        "completedAt":      datetime.utcnow().isoformat(),
        "notes":            "Challenger registered by retrain.py",
    }
    requests.post(
        f"{API_BASE}/api/ml/training-run",
        json=payload, headers=_headers(), timeout=10
    ).raise_for_status()


def main(dry_run: bool = False) -> None:
    logging.basicConfig(level=logging.INFO)
    started_at = datetime.utcnow().isoformat()

    log.info("retrain.py starting — API_BASE=%s", API_BASE)

    feedback_df  = fetch_feedback()
    features_df  = load_features_from_db()
    enriched     = merge_feedback(features_df, feedback_df)

    if dry_run:
        log.info("Dry run — data looks good (%d rows). Exiting without training.", len(enriched))
        return

    pipeline, metrics, feature_importance, n_rows = train(enriched)
    version  = get_next_version()
    model_id = register(pipeline, version, metrics, feature_importance,
                        n_rows, len(feedback_df))
    record_run(model_id, n_rows, len(feedback_df), started_at)

    log.info("Challenger %s registered (id=%s). F1=%.3f",
             version, model_id[:8], metrics["f1"])


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--dry-run", action="store_true")
    args = parser.parse_args()
    main(dry_run=args.dry_run)
```

```python
def compute_tank_features(site_id: str, as_of_date, conn) -> dict:
    """
    Compute tank-level features for a site.
    Returns zeros/nulls gracefully if fact_tank_telemetry is empty.
    """
    with conn.cursor() as cur:
        cur.execute("""
            SELECT
                COUNT(*) FILTER (WHERE overfill_flag = TRUE) AS overfill_events_7d,
                AVG(tank_level_pct) AS avg_tank_level_7d
            FROM fact_tank_telemetry
            WHERE site_id = %s
              AND reading_timestamp >= %s::date - INTERVAL '7 days'
              AND reading_timestamp <  %s::date
        """, (site_id, as_of_date, as_of_date))
        row = cur.fetchone()
    return {
        "tank_overfill_events_7d": int(row[0] or 0),
        "avg_tank_level_pct_7d":   float(row[1] or 0.0),
    }
```
